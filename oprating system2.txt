let count=0
echo "Please type 'finish' to quit this demo."

# Prompt the user with "Say something > " and
# read the value into variable "x"
while read -p "Say something > " x
do
    if [ "${x}" == "finish" ]
    then
        break
    fi

    if [ "${x}" == "abort" ]
    then
        exit
    fi

    # increment the counter
    let count++

    # if we say "silent", then we count the word
    # but do not display it:
    if [ "${x}" == "silent" ]
    then
        continue
    fi

    echo "you said : '${x}'"
    echo "you have said ${count} things."
done

echo "All done. you have said ${count} things to me."




6 whether string contain unique words or not

vi won.sh


echo "Enter the string"
read str
i=0

# Loop through each character of the string
while [ $i -lt ${#str} ]
do
    # Set j to the next character's index
    j=$((i+1))
    
    # Compare current character at index i with characters at index j
    while [ $j -lt ${#str} ]
    do
        # String slicing ${str:offset:length}
        if [ "${str:$i:1}" = "${str:$j:1}" ]
        then
            echo "words in a string are NOT unique"
            exit
        fi
        let j++
    done
    let i++
done

echo "Words in a string are unique"

while :
do
    echo " Menu "
    echo "1. Copy a file"
    echo "2. Rename a file"
    echo "3. remove a file"
    echo "4. exit"
    echo "enter your choice : "
    read ch

    case $ch in
        1)
            echo "Enter file name to copy : "
            read f1
            echo "Enter new file name (copy name) : "
            read f2
            if [ -f $f1 ]
            then
                cp $f1 $f2
                echo "$f1 is copied to $f2"
            else
                echo "$f1 does not exist"
            fi
            ;;

        2)
            echo "Enter filename to rename: "
            read f1
            echo "Enter new file name: "
            read f2
            if [ -f $f1 ]
            then
                mv $f1 $f2
                echo "$f1 is renamed to $f2"
            else
                echo "$f1 does not exist"
            fi
            ;;

        3)
            echo "Enter file name to remove : "
            read f1
            if [ -f $f1 ]
            then
                rm -i $f1
                echo "$f1 is removed"
            else
                echo "$f1 does not exist"
            fi
            ;;

        4)
            echo "Exit.."
            exit
            ;;

        *)
            echo "Invalid choice"
            ;;
    esac
done
